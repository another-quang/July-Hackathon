```csharp
namespace Team1Abr.Core;

/// <summary>
/// Composes a short, plain-language summary of a business record.
///
/// Offline and deterministic: the text is generated from the record's own fields —
/// no network calls, no live model. This honours the hackathon rule to use fixtures
/// only and keeps the output unit-testable. The UI labels it as automatically
/// generated so its provenance is clear to the reader.
/// </summary>
public static class Summary
{
    public static string Generate(BusinessRecord record)
    {
        var statusClause = string.Equals(record.AbnStatus, "Active", StringComparison.OrdinalIgnoreCase)
            ? "Its ABN is currently active"
            : "Its ABN has been cancelled, so it may no longer be trading";

        var gstClause = record.GstRegistered
            ? " and it is registered for GST."
            : ". It is not registered for GST.";

        return $"{record.EntityName} is a {record.EntityType} based in {record.State} " +
               $"(postcode {record.Postcode}). {statusClause}{gstClause}";
    }
}
```

In `src/Core/Lookup.cs`, add these three methods inside the `Lookup` class, immediately after the existing `LookupAbn` method (around line 39):

```csharp
    /// <summary>All business records from the local sample data.</summary>
    public static IReadOnlyList<BusinessRecord> AllRecords() => Records;

    /// <summary>
    /// Records whose state matches <paramref name="state"/>. A null/empty value or
    /// "All" (case-insensitive) returns every record. Matching is case-insensitive.
    /// </summary>
    public static IReadOnlyList<BusinessRecord> FilterByState(string? state)
    {
        if (string.IsNullOrWhiteSpace(state) ||
            string.Equals(state, "All", StringComparison.OrdinalIgnoreCase))
        {
            return Records;
        }

        return Records
            .Where(record => string.Equals(record.State, state, StringComparison.OrdinalIgnoreCase))
            .ToList();
    }

    /// <summary>The sorted, unique set of states present in the sample data.</summary>
    public static IReadOnlyList<string> DistinctStates() =>
        Records
            .Select(record => record.State)
            .Distinct()
            .OrderBy(state => state, StringComparer.Ordinal)
            .ToList();
```

Note: `System.Linq` is available via the project's `ImplicitUsings`, so no extra `using` is required.

```csharp
    private async Task HandleSubmit()
    {
        _copyMessage = string.Empty;
        _isLoading = true;

        // Brief, deliberate delay so the accessible loading state is perceivable.
        // Lookups are in-memory and otherwise instant.
        await Task.Delay(300);

        var result = Lookup.SearchAbn(_abnInput);
        _isLoading = false;

        if (result.Status == LookupStatus.Error)
        {
            _state = LookupStatus.Error;
            _errorMessage = result.Reason;
            return;
        }

        AddRecentSearch(result.NormalisedAbn);

        if (result.Status == LookupStatus.NotFound)
        {
            _state = LookupStatus.NotFound;
            _notFoundAbn = Abn.Format(result.NormalisedAbn);
            _record = null;
            return;
        }

        _record = result.Record;
        _state = LookupStatus.Found;
    }
```

```csharp
    private async Task HandleRecentSearch(string normalisedAbn)
    {
        _copyMessage = string.Empty;
        _abnInput = Abn.Format(normalisedAbn);
        _isLoading = true;

        await Task.Delay(300);

        var record = Lookup.LookupAbn(normalisedAbn);
        _isLoading = false;

        if (record is null)
        {
            _state = LookupStatus.NotFound;
            _notFoundAbn = Abn.Format(normalisedAbn);
            _record = null;
            return;
        }

        _record = record;
        _state = LookupStatus.Found;
    }
```

Note: Blazor re-renders when the async handler yields at the first `await`, so setting `_isLoading = true` before the delay shows the indicator without an explicit `StateHasChanged()`.
